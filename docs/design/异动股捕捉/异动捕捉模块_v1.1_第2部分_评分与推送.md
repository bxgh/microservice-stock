## E3 · 综合评分与可解释性

### E3-S1 评分公式(增强版)

```text
composite_score = α · raw_score
                + β · score_l3_capital
                + γ · score_l4_emotion
                + ε · score_user_pref
                - ζ · score_dedup_penalty
                + μ · (resonance_level - 1) · 15        ← 共振加成
                - ν · counter_signal_score / 100 · 20   ← 反向抑制

最终乘以市场状态可信度系数:
composite_score *= ads_market_state_daily.signal_reliability

权重(初始,可热配置):
  α = 0.40    raw_score
  β = 0.25    L3 资金
  γ = 0.15    L4 情绪
  ε = 0.10    个人偏好
  ζ = 0.10    重复跟踪压制
  μ = 0.15    共振等级加成系数
  ν = 0.20    反向信号抑制系数

约束:
  - 各子项打分 ∈ [0, 100]
  - composite_score 计算后 LEAST(100, GREATEST(0, x)) clamp
```

### E3-S2 子项打分细则

详见 v1.0 文档 E3-S2,本版保持一致。重点说明 v1.1 新增项:

#### E3-S2-T6 共振加成

```sql
-- resonance_level=1 加 0,=2 加 15,=3 加 30,=4 加 45,=5 加 60(再乘 μ=0.15)
-- 实际公式: + μ · (resonance_level - 1) · 15
```

#### E3-S2-T7 反向抑制

```sql
-- counter_signal_score=0 不扣,=100 扣 20(再乘 ν=0.20)
-- 实际公式: - ν · (counter_signal_score / 100) · 20
```

### E3-S3 综合评分计算 SQL

```sql
-- 文件: sql/anomaly/06_composite_score.sql

UPDATE ads_l8_unified_signal s
JOIN dim_anomaly_score_weight w_a ON w_a.weight_key = 'alpha_raw_score'     AND w_a.is_active = 1
JOIN dim_anomaly_score_weight w_b ON w_b.weight_key = 'beta_l3_capital'     AND w_b.is_active = 1
JOIN dim_anomaly_score_weight w_g ON w_g.weight_key = 'gamma_l4_emotion'    AND w_g.is_active = 1
JOIN dim_anomaly_score_weight w_e ON w_e.weight_key = 'epsilon_user_pref'   AND w_e.is_active = 1
JOIN dim_anomaly_score_weight w_z ON w_z.weight_key = 'zeta_dedup_penalty'  AND w_z.is_active = 1
JOIN dim_anomaly_score_weight w_m ON w_m.weight_key = 'mu_resonance_boost'  AND w_m.is_active = 1
JOIN dim_anomaly_score_weight w_n ON w_n.weight_key = 'nu_counter_penalty'  AND w_n.is_active = 1
LEFT JOIN ads_market_state_daily ms ON ms.trade_date = s.trade_date
SET s.composite_score = LEAST(100, GREATEST(0,
    (
        s.raw_score        * w_a.weight_value
      + s.score_l3_capital * w_b.weight_value
      + s.score_l4_emotion * w_g.weight_value
      + s.score_user_pref  * w_e.weight_value
      - s.score_dedup_pen  * w_z.weight_value
      + (IFNULL(s.resonance_level, 1) - 1) * 15 * w_m.weight_value
      - (IFNULL(s.counter_signal_score, 0) / 100.0) * 20 * w_n.weight_value
    ) * IFNULL(ms.signal_reliability, 1.00)
))
WHERE s.trade_date = @td;
```

```python
        }.get(signal.get('signal_subtype'), '处于启动前征兆')

            'false_breakout':    '突破失败,警惕假突破',
            'high_vol_topping':  '高位巨量出货特征',
            'leader_topping':    '板块龙头见顶迹象'
        }.get(signal.get('signal_subtype'), '触发陷阱信号')
        parts.append(f"{name} {sub_text}")
    
    # 第二段:板块情况
    if 'sector_leader' in tags:
        parts.append(f"是 {signal['industry_sw1']} 板块当日唯一首板")
    elif 'mainline_resonance' in tags:
        parts.append(f"所属 {signal['industry_sw1']} 板块强势")
    
    # 第三段:资金情况
    if 'main_inflow_strong' in tags:
        parts.append(f"主力资金净流入 {signal['main_net_inflow']/1e8:.2f} 亿")
    if 'lhb_inst_buy' in tags:
        parts.append("龙虎榜机构席位净买入")
    
    # 第四段:形态情况
    pattern_map = {
        'pre_zt_consolidation':   '涨停前 10 日横盘整理',
        'pre_zt_low_volume':      '涨停前缩量蓄势',
        'pre_zt_already_pulled':  '⚠️ 涨停前已加速,需警惕',
        'pre_zt_pullback':        '涨停前有过回调',
        'pre_zt_first_recovery':  '首次站稳年线,长期反转信号',
        'breakout_box_60d':       '突破 60 日箱体',
    }
    for tag, text in pattern_map.items():
        if tag in tags:
            parts.append(text)
            break
    
    # 第五段:外部
    if 'sox_up_overnight' in tags:
        parts.append("隔夜费城半导体指数上涨")
    
    # 第六段:共振等级
    res_level = signal.get('resonance_level', 0)
    if res_level >= 4:
        hit_dims = [d for d, info in resonance_dims.items() if info.get('hit')]
        parts.append(f"共振等级 L{res_level}({'、'.join(hit_dims)}多维印证)")
    
    # 第七段:反向信号(警示)
    if counter:
        warnings = []
        if counter.get('fundamental'):
            warnings.append('⚠️ 有基本面利空')
        if counter.get('supply'):
            warnings.append('⚠️ 临近解禁')
        if counter.get('sentiment'):
            warnings.append('⚠️ 板块情绪过热')
        if warnings:
            parts.append('、'.join(warnings))
    
    return '。'.join(parts) + '。'
```

## E4 · Top 10 推送规则

### E4-S1 推送生成总流程

```text
1. 读取用户当前激活的 Profile
2. 加载当日所有信号(WHERE default_visible=1 + 应用 Profile 的 exclude 规则)
3. 应用 Profile 的 prefer_tags 加权
4. L5 共振必入(最多 3 条)
5. 各池按 quota 取(strong/early/trap)
6. 陷阱保底(如不足 trap_min 从 trap 池补 1)
7. 跨股票去重
8. 缺额从未选中的 strong/early 池填补
9. 排序赋 rank_no,生成 headline 和 key_features
10. 写入 app_anomaly_top10_daily
```

### E4-S2 Python 实现

```python
# 文件: scripts/anomaly/top10.py

def generate_top10(trade_date: str, user_id: int = 1):
    eng = get_engine()
    
    # 1. 读取激活 Profile
    profile = get_active_profile(user_id, eng)
    rules = json.loads(profile['rules_json'])
    
    # 2. 读取配置
    cfg = pd.read_sql("""
        SELECT weight_key, weight_value 
        FROM dim_anomaly_score_weight 
        WHERE is_active = 1
    """, eng).set_index('weight_key')['weight_value'].to_dict()
    
    quotas = rules.get('pool_quotas', {
        'strong': int(cfg['pool_strong_quota']),
        'early':  int(cfg['pool_early_quota']),
        'trap':   int(cfg['pool_trap_quota'])
    })
    trap_min = rules.get('trap_min', int(cfg['pool_trap_min']))
    l5_max = int(cfg['l5_must_max'])
    top_n = rules.get('top_n', 10)
    min_resonance = rules.get('min_resonance', 1)
    
    # 3. 拉取候选
    df = load_candidates(trade_date, user_id, rules, eng)
    df = df[df['resonance_level'] >= min_resonance]
    
    # 4. 应用 Profile 偏好加权
    if rules.get('prefer_tags'):
        boost = rules.get('boost_factor', 1.5)
        df['adjusted_score'] = df.apply(
            lambda r: r['composite_score'] * boost
                      if any(t in r['tags'] for t in rules['prefer_tags'])
                      else r['composite_score'],
            axis=1
        )
    else:
        df['adjusted_score'] = df['composite_score']
    
    # 5. L5 共振必入
    selected = []
    l5_df = df[df['resonance_level'] == 5].sort_values('adjusted_score', ascending=False).head(l5_max)
    if not l5_df.empty:
        l5_df = l5_df.copy()
        l5_df['quota_slot'] = 'quota_l5_must'
        selected.append(l5_df)
    
    selected_codes = set(l5_df['ts_code']) if not l5_df.empty else set()
    
    # 6. 各池配额(去掉已被 L5 选中的)
    for pool, q in quotas.items():
        sub = df[(df['pool_type'] == pool) & (~df['ts_code'].isin(selected_codes))]
        sub = sub.sort_values('adjusted_score', ascending=False).head(q).copy()
        sub['quota_slot'] = f'quota_{pool}'
        selected.append(sub)
        selected_codes |= set(sub['ts_code'])
    
    selected_df = pd.concat(selected) if selected else pd.DataFrame()
    
    # 7. 陷阱保底
    trap_count = (selected_df['pool_type'] == 'trap').sum()
    if trap_count < trap_min:
        trap_extra = df[(df['pool_type'] == 'trap') 
                        & (~df['ts_code'].isin(selected_codes))].head(trap_min - trap_count).copy()
        if not trap_extra.empty:
            trap_extra['quota_slot'] = 'quota_trap'
            selected_df = pd.concat([selected_df, trap_extra])
            selected_codes |= set(trap_extra['ts_code'])
    
    # 8. 跨股票去重(防止同股多池命中)
    selected_df = selected_df.sort_values('adjusted_score', ascending=False)
    selected_df = selected_df.drop_duplicates(subset='ts_code', keep='first')
    
    # 9. 填补缺额
    shortage = top_n - len(selected_df)
    if shortage > 0:
        candidates = df[(~df['ts_code'].isin(selected_codes))
                        & df['pool_type'].isin(['strong', 'early'])]
        candidates = candidates.sort_values('adjusted_score', ascending=False).head(shortage).copy()
        candidates['quota_slot'] = 'quota_filled'
        selected_df = pd.concat([selected_df, candidates])
    
    # 10. 排序并赋 rank_no
    final = selected_df.sort_values('adjusted_score', ascending=False).head(top_n).reset_index(drop=True)
    final['rank_no'] = range(1, len(final) + 1)
    final['profile_code'] = profile['profile_code']
    final['user_id'] = user_id
    final['signal_id'] = final['id']  # 关联到 ads_l8_unified_signal
    
    # 11. 生成 headline 和 key_features
    final['headline'] = final.apply(generate_headline, axis=1)
    final['key_features'] = final.apply(generate_key_features, axis=1)
    
    # 12. 写入数据库
    final[['user_id', 'trade_date', 'rank_no', 'signal_id', 'ts_code', 'name',
           'industry_sw1', 'pool_type', 'signal_type', 'signal_subtype',
           'composite_score', 'resonance_level', 'quota_slot', 'profile_code',
           'headline', 'key_features']].to_sql(
        'app_anomaly_top10_daily', eng, if_exists='append', index=False
    )
```

### E4-S3 验收标准

> **Given** 用户激活默认 Profile,当日有 5 条 L5 共振信号  
> **When** Top 10 生成  
> **Then** 推送清单中 L5 信号最多 3 条(`quota_slot='quota_l5_must'`),其余按各池配额选取

> **Given** 用户切换到 `profile_short_term`(短线接力派)  
> **When** 重新生成 Top 10  
> **Then** 推送清单中含 `sector_leader` 或 `relay_candidate` 标签的比例 ≥ 70%
